  2  (
  3      lv_state IN BB_TAX.STATE%TYPE,
  4      subtotal IN NUMBER,
  5      tax OUT NUMBER
  6  )
  7  IS

  9

SQL> -- 5-6: Returning Order Status Information
SQL> CREATE OR REPLACE PROCEDURE STATUS_SP
  2  (
  3      lv_basketid IN BB_BASKETSTATUS.IDBASKET%TYPE,
  4      lv_status OUT VARCHAR
  5  )
  6  IS
  7      status_num BB_BASKETSTATUS.IDSTATUS%TYPE;
  8  BEGIN
  9      SELECT IDSTAGE 
 10          INTO status_num
 11          FROM BB_BASKETSTATUS
 12          WHERE DTSTAGE = (
 13                  SELECT MAX(DTSTAGE) FROM BB_BASKETSTATUS
 14                  WHERE IDBASKET = lv_basketid
 15              );
 16      CASE status_num
 17          WHEN 1 THEN
 18              lv_status := 'Submitted and recieved';
 19          WHEN 2 THEN
 20              lv_status := 'Confirmed, processed, sent to shipping';
 21          WHEN 3 THEN
 22              lv_status := 'Shipped';
 23          WHEN 4 THEN
 24              lv_status := 'Cancelled';
 25          WHEN 5 THEN
 26              lv_status := 'Back-ordered';
 27          ELSE
 28              lv_status := 'No status is available';
 29      END CASE;
 30  EXCEPTION
 31      WHEN NO_DATA_FOUND THEN
 32          lv_status := 'No status is available';
 33  END;

SQL> CREATE OR REPLACE PROCEDURE PROMO_SHIP_SP
  2  (
  3      lv_cutoff_date IN BB_BASKET.DTCREATED%TYPE,
  4      lv_month IN BB_PROMOLIST.MONTH%TYPE,
  5      lv_year IN BB_PROMOLIST.YEAR%TYPE,
  6      lv_promo_flag IN BB_PROMOLIST.PROMO_FLAG%TYPE
  7  )
  8  IS
  9      CURSOR cur_shoppers IS
 10          SELECT DISTINCT IDSHOPPER FROM BB_BASKET b
 11              WHERE NOT EXISTS (
 12                  SELECT 1
 13                  FROM BB_BASKET
 14                  WHERE IDSHOPPER = b.IDSHOPPER
 15                      AND DTCREATED >= lv_cutoff_date
 16              );
 17  
 18      shopper BB_BASKET.IDSHOPPER%TYPE;
 19  BEGIN
 20      FOR shopper in cur_shoppers LOOP
 21          INSERT INTO BB_PROMOLIST (idshopper, month, year, promo_flag, used)
 22              VALUES(shopper.IDSHOPPER, lv_month, lv_year, lv_promo_flag, 'N');
 23      END LOOP;
 24      COMMIT;
 25  END;
